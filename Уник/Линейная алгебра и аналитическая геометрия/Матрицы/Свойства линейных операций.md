1. A + B = B + A - коммутативность
2. ( A + B ) + C = A + ( B + C ) - ассоциативность
3. λ ⋅ ( A + B ) = λ ⋅ A + λ ⋅ B - дистрибутивность
4. ƎO : A + O = A, где O - нулевая матрица
5. ( λ + μ ) ⋅ A = λ ⋅ A + μ ⋅ A - дистрибутивность
6. λ ⋅ ( μ ⋅ A ) = ( λ ⋅ μ ) ⋅ A - ассоциативность
7. 1 ⋅ A = A - унитарность
8. Ǝ(-A) - противоположность; A + (-A) = 0
